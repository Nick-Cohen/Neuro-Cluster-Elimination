```python
def validate_nn_section(config):
    """Validate and flatten nn_config fields if present in experiment config.

    If the experiment config contains nn-specific sections (inference, nn, training,
    sampling, backward, output), extract them and validate via config_schema.

    Args:
        config: Experiment config dict (may contain nn sections at top level)

    Returns:
        Dict with nn_config fields flattened (if nested sections were present)
        or the original config unchanged (if no nested sections found)
    """
    nn_sections = {'inference', 'nn', 'training', 'sampling', 'backward', 'output'}
    has_nn_sections = any(key in config for key in nn_sections)

    if not has_nn_sections:
        return config

    # Extract nn sections into a separate dict
    nn_config = {}
    remaining = {}
    for key, value in config.items():
        if key in nn_sections:
            nn_config[key] = value
        else:
            remaining[key] = value

    # Validate and flatten the nn config
    flat_nn = flatten_nn_config(nn_config)

    # Merge back: experiment-level fields + flattened nn fields
    result = dict(remaining)
    result['nn_config_flat'] = flat_nn  # Store for experiment runner to use

    return result
```
